# INDEPENDENT KAGGLE SESSION — ONE EXPERIMENT / ONE CELL
# Experiment : C12_Caltech101_R18_10ep
# Dataset    : caltech101
# Backbone   : resnet18
# Epochs     : 10
# Batch size : 64

# Canonical CNN family used by the paper:
#   DT1D, Full, Linear, BitFit, SSF, Conv-r4, BAM,
#   Residual Adapter, LoRA-Conv, Side-Tuning.
#
# This cell is independent: it clones the repository, applies the
# validated Side-Tuning + LR-search fixes, validates, smoke-tests,
# downloads data/weights, trains, and packages results by itself.
# HARD SESSION GUARD: at 11h20m (40,800 s) from cell start, active
# training process groups are terminated and two ZIPs are exported:
#   *_results.zip            = resumable full state (including partial work)
#   *_DONE_ONLY_results.zip  = only methods complete for ALL requested seeds
# ============================================================

pack_results() {
  [[ -d "$STATE" ]] || return 0
  python - <<'PYZIP'
import os, zipfile
from pathlib import Path
root=Path(os.environ["STATE"]); dst=Path(os.environ["ZIP"])
tmp=dst.with_name(dst.name+f".tmp.{os.getpid()}")
tmp.unlink(missing_ok=True)
with zipfile.ZipFile(tmp,"w",zipfile.ZIP_DEFLATED,compresslevel=6) as z:
    for p in root.rglob("*"):
        if p.is_file() and p.suffix.lower() not in {".pth",".pt",".ckpt"}:
            z.write(p,p.relative_to(root.parent))
tmp.replace(dst)
print("PACKED RESUMABLE STATE:",dst)
PYZIP
  ls -lh "$ZIP" || true
}

# At the 11h20 cutoff, this second archive contains ONLY methods that
# have final test_summary.json for every requested seed. This prevents
# a partially trained method from being mistaken for a completed result.
pack_done_results() {
  mkdir -p "$STATE"
  python - <<'PYDONE'
import json, os, zipfile
from pathlib import Path
state=Path(os.environ["STATE"])
dst=Path(os.environ["DONE_ZIP"])
target=os.environ["EXPERIMENT_ID"]
methods=[x for x in os.environ["METHODS_CSV"].split(",") if x]
seeds=[int(x) for x in os.environ["SEEDS_CSV"].split(",") if x]
start=int(os.environ["CELL_START_TS"])
now=int(__import__('time').time())
status={}
complete=[]
for method in methods:
    root=state/"outputs"/method/target/method
    ok=all((root/f"seed_{seed}"/"test_summary.json").is_file() for seed in seeds)
    status[method]={"complete":bool(ok),"method_root":str(root)}
    if ok: complete.append((method,root))

tmp=dst.with_name(dst.name+f".tmp.{os.getpid()}")
tmp.unlink(missing_ok=True)
with zipfile.ZipFile(tmp,"w",zipfile.ZIP_DEFLATED,compresslevel=6) as z:
    manifest={
        "job":os.environ["JOB"],
        "experiment_id":target,
        "requested_methods":methods,
        "requested_seeds":seeds,
        "elapsed_seconds":now-start,
        "budget_seconds":int(os.environ["CELL_BUDGET_SECONDS"]),
        "complete_methods":[m for m,_ in complete],
        "status":status,
    }
    z.writestr("DONE_ONLY_MANIFEST.json",json.dumps(manifest,indent=2)+"\n")
    cfg=Path(os.environ["CONFIG"])
    if cfg.is_file(): z.write(cfg,"experiment.yaml")
    for method,root in complete:
        for p in root.rglob("*"):
            if p.is_file() and p.suffix.lower() not in {".pth",".pt",".ckpt"}:
                z.write(p,Path(target)/method/p.relative_to(root))
        log=state/"logs"/f"{method}.log"
        if log.is_file(): z.write(log,Path("logs")/log.name)
tmp.replace(dst)
print("PACKED DONE-ONLY:",dst)
print("COMPLETE METHODS:",[m for m,_ in complete])
PYDONE
  ls -lh "$DONE_ZIP" || true
}

pack_all_exports() {
  pack_results || true
  pack_done_results || true
}

stop_active_training_groups() {
  [[ -f "$ACTIVE_PGIDS_FILE" ]] || return 0
  mapfile -t _pgids < "$ACTIVE_PGIDS_FILE" || true
  for pgid in "${_pgids[@]:-}"; do
    [[ "$pgid" =~ ^[0-9]+$ ]] || continue
    kill -TERM -- "-$pgid" 2>/dev/null || true
  done
  sleep 20
  for pgid in "${_pgids[@]:-}"; do
    [[ "$pgid" =~ ^[0-9]+$ ]] || continue
    kill -KILL -- "-$pgid" 2>/dev/null || true
  done
}

watchdog_11h20() {
  local now remaining
  now=$(date +%s)
  remaining=$((CELL_DEADLINE_TS - now))
  if (( remaining > 0 )); then sleep "$remaining"; fi
  mkdir -p "$STATE"
  {
    echo "11h20 wall-clock limit reached"
    echo "cell_start_epoch=$CELL_START_TS"
    echo "deadline_epoch=$CELL_DEADLINE_TS"
    echo "reached_epoch=$(date +%s)"
  } > "$TIMEOUT_FLAG"
  echo
  echo "============================================================"
  echo "11h20 LIMIT REACHED — stopping active training safely"
  echo "Completed methods will be exported immediately."
  echo "============================================================"
  stop_active_training_groups
  # Export immediately from the watchdog itself. ZIP creation is atomic,
  # so a simultaneous end-of-wave checkpoint cannot leave a corrupt archive.
  pack_all_exports || true
  echo "11h20 AUTO-EXPORT COMPLETE: $DONE_ZIP"
  echo "11h20 RESUMABLE STATE:     $ZIP"
}

on_exit() {
  local rc=$?
  trap - EXIT
  if [[ -n "${WATCHDOG_PID:-}" ]]; then kill "$WATCHDOG_PID" 2>/dev/null || true; fi
  pack_all_exports || true
  exit "$rc"
}
trap on_exit EXIT

start=t.index("def select_shared_lr("); end=t.index("\n\ndef candidate_for(",start)

    expected={(int(s),float(lr)) for s in seeds for lr in lr_candidates}
    observed={(int(r["seed"]),float(r["lr"])) for r in candidate_rows}
    if observed != expected or len(candidate_rows) != len(expected):
        raise RuntimeError(f"Fair LR search incomplete/duplicated: missing={sorted(expected-observed)} extra={sorted(observed-expected)}")
    def good(r):
        try: v=float(r.get("best_val_acc1",float("nan")))
        except (TypeError,ValueError): return False
        return int(r.get("return_code",1)) == 0 and math.isfinite(v)
    def numerical(r):
        if good(r): return False
        rc=int(r.get("return_code",1))
        if rc == 0: return True
        lp=Path(str(r.get("output_dir","")))/"stdout.log"
        log=lp.read_text(encoding="utf-8",errors="ignore") if lp.is_file() else ""
        if "Loss is not finite" in log or "FloatingPointError: Loss is not finite" in log:
            return True
        raise RuntimeError(f"Non-numerical candidate failure seed={r.get('seed')} lr={r.get('lr')} rc={rc}; inspect {lp}")
    invalid=[r for r in candidate_rows if numerical(r)]
    aggs=[]; eligible=[]
    for lr in lr_candidates:
        rows=sorted([r for r in candidate_rows if float(r["lr"])==float(lr)],key=lambda r:int(r["seed"]))
        valid=all(good(r) for r in rows)
        vals=[float(r["best_val_acc1"]) for r in rows] if valid else []
        a={"lr":float(lr),"valid":valid,
           "mean_best_val_acc1":float(statistics.mean(vals)) if valid else None,
           "std_best_val_acc1":float(statistics.stdev(vals)) if valid and len(vals)>1 else (0.0 if valid else None),
           "per_seed_best_val_acc1":{str(int(r["seed"])):(float(r["best_val_acc1"]) if good(r) else None) for r in rows},
           "per_seed_best_epoch":{str(int(r["seed"])):int(r.get("best_epoch",-1)) for r in rows},
           "per_seed_return_code":{str(int(r["seed"])):int(r.get("return_code",1)) for r in rows}}
        aggs.append(a)
        if valid: eligible.append(a)
    if not eligible: raise RuntimeError("No LR completed successfully for every requested seed.")
    sel=max(eligible,key=lambda a:(float(a["mean_best_val_acc1"]),-float(a["lr"])))
    return {"selection_scope":"method_across_seeds","aggregation":"mean_best_val_acc1","tie_break":"lower_lr",
            "failed_candidate_policy":"attempt_all_mark_numerical_invalid","all_declared_candidates_attempted":True,
            "seeds":[int(s) for s in seeds],"lr_candidates":[float(lr) for lr in lr_candidates],
            "numerical_invalid_candidates":[{"seed":int(r["seed"]),"lr":float(r["lr"]),"return_code":int(r.get("return_code",1))} for r in invalid],
            "candidates":aggs,"selected_lr":float(sel["lr"]),
            "selected_mean_best_val_acc1":float(sel["mean_best_val_acc1"])}
"""
t=t[:start]+new_func+t[end:]

launch_one() {
  local method="$1" device="$2" out="$STATE/outputs/$1" log="$STATE/logs/$1.log"
  mkdir -p "$out"
  # setsid gives each method its own process group so the 11h20 watchdog
  # can stop Python + tee + descendants without touching the notebook shell.
  setsid bash -c '
    set -Eeuo pipefail
    method="$1"; device="$2"; out="$3"; log="$4"
    python "$REPO/tools/run_experiment.py" "$CONFIG" \
      --methods "$method" --seeds "$SEEDS_CSV" \
      --output-root "$out" --data-path "$DATA" --device "$device" --skip-if-complete \
      2>&1 | tee "$log"
  ' _ "$method" "$device" "$out" "$log" &
  LAST_LAUNCH_PID=$!
}

for ((i=0; i<${#METHODS[@]}; i+=WAVE)); do
  now=$(date +%s)
  if (( now >= CELL_DEADLINE_TS )); then
    [[ -f "$TIMEOUT_FLAG" ]] || echo "11h20 wall-clock limit reached before next wave" > "$TIMEOUT_FLAG"
    pack_all_exports
    echo "STOPPED AT 11h20. DONE ZIP: $DONE_ZIP"
    echo "RESUME ZIP: $ZIP"
    if [[ -n "${WATCHDOG_PID:-}" ]]; then kill "$WATCHDOG_PID" 2>/dev/null || true; fi
    trap - EXIT
    exit 124
  fi
  echo "Remaining cell budget: $((CELL_DEADLINE_TS-now)) seconds"

  pids=(); names=()
  for ((j=0; j<WAVE && i+j<${#METHODS[@]}; j++)); do
    m="${METHODS[i+j]}"
    launch_one "$m" "cuda:$j"
    pids+=("$LAST_LAUNCH_PID"); names+=("$m")
  done
  printf '%s\n' "${pids[@]}" > "$ACTIVE_PGIDS_FILE"

  set +e; wave_rc=0
  for k in "${!pids[@]}"; do
    wait "${pids[$k]}"; rc=$?
    if [[ "$rc" -ne 0 ]]; then echo "STOP/FAIL ${names[$k]} rc=$rc"; wave_rc=$rc; fi
  done
  set -e
  : > "$ACTIVE_PGIDS_FILE"

  # Always checkpoint the session after each wave.
  pack_all_exports

  # The watchdog deliberately kills the active groups at 11h20. In that
  # case export completed methods and exit cleanly with a distinct code.
  if [[ -f "$TIMEOUT_FLAG" ]]; then
    echo "11h20 guard activated. No new methods will be launched."
    echo "DONE ZIP: $DONE_ZIP"
    echo "RESUME ZIP: $ZIP"
    if [[ -n "${WATCHDOG_PID:-}" ]]; then kill "$WATCHDOG_PID" 2>/dev/null || true; fi
    trap - EXIT
    exit 124
  fi

  [[ "$wave_rc" -eq 0 ]] || exit "$wave_rc"
done
