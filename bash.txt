pack_results () {

python - <<'PY'
import os
import json
import zipfile
from pathlib import Path

target = os.environ["TARGET"]

methods = [
    x for x in os.environ["METHODS"].split(",")
    if x
]

seeds = [
    int(x) for x in os.environ["SEEDS"].split(",")
    if x
]

final_root = Path(os.environ["FINAL_ROOT"])

result_zip = Path(os.environ["RESULT_ZIP"])
done_zip   = Path(os.environ["DONE_ZIP"])


def is_complete(method):

    root = final_root / target / method

    if not (root / "lr_selection_summary.json").is_file():
        return False

    for seed in seeds:

        p = root / f"seed_{seed}" / "test_summary.json"

        if not p.is_file():
            return False

    return True


complete = [
    m for m in methods
    if is_complete(m)
]


status = {

    "session": os.environ["SESSION_ID"],

    "target": target,

    "methods": methods,

    "seeds": seeds,

    "complete_methods": complete,

    "missing_methods": [
        m for m in methods
        if m not in complete
    ],
}


final_root.mkdir(
    parents=True,
    exist_ok=True
)


(
    final_root /
    "EXPORT_STATUS.json"
).write_text(
    json.dumps(
        status,
        indent=2
    )
    + "\n"
)


for p in [
    result_zip,
    done_zip
]:

    if p.exists():
        p.unlink()


# ------------------------------------------------------------
# FULL ZIP
# ------------------------------------------------------------

with zipfile.ZipFile(
    result_zip,
    "w",
    zipfile.ZIP_DEFLATED,
    compresslevel=6
) as z:

    for root in [

        final_root,

        Path(os.environ["OUT_A"]),

        Path(os.environ["OUT_B"]),
    ]:

        if not root.exists():
            continue

        for p in root.rglob("*"):

            if not p.is_file():
                continue

            # no checkpoints in export
            if p.suffix.lower() in {
                ".pth",
                ".pt",
                ".ckpt"
            }:
                continue

            z.write(
                p,
                Path(root.name) /
                p.relative_to(root)
            )


# ------------------------------------------------------------
# DONE-ONLY ZIP
# ------------------------------------------------------------

with zipfile.ZipFile(
    done_zip,
    "w",
    zipfile.ZIP_DEFLATED,
    compresslevel=6
) as z:

    # status
    status_file = (
        final_root /
        "EXPORT_STATUS.json"
    )

    if status_file.exists():

        z.write(
            status_file,
            Path(final_root.name) /
            status_file.name
        )


    # aggregation
    agg = (
        final_root /
        "aggregated"
    )

    if agg.exists():

        for p in agg.rglob("*"):

            if p.is_file():

                z.write(
                    p,
                    Path(final_root.name) /
                    p.relative_to(final_root)
                )


    # only complete methods
    for method in complete:

        root = (
            final_root /
            target /
            method
        )

        for p in root.rglob("*"):

            if not p.is_file():
                continue

            if p.suffix.lower() in {
                ".pth",
                ".pt",
                ".ckpt"
            }:
                continue

            z.write(
                p,
                Path(final_root.name) /
                target /
                method /
                p.relative_to(root)
            )


print(
    json.dumps(
        status,
        indent=2
    )
)

print(
    "\nFULL ZIP:",
    result_zip
)

print(
    "DONE ZIP:",
    done_zip
)
PY

ls -lh \
    "$RESULT_ZIP" \
    "$DONE_ZIP" \
    2>/dev/null || true
}
