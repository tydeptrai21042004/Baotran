pack_results() {
  [[ -d "$STATE" ]] || return 0
  python - <<'PYZIP'
import os, zipfile
from pathlib import Path
root=Path(os.environ["STATE"]); dst=Path(os.environ["ZIP"])
tmp=dst.with_name(dst.name+f".tmp.{os.getpid()}")
tmp.unlink(missing_ok=True)
with zipfile.ZipFile(tmp,"w",zipfile.ZIP_DEFLATED,compresslevel=6) as z:
    for p in root.rglob("*"):
        if p.is_file() and p.suffix.lower() not in {".pth",".pt",".ckpt"}:
            z.write(p,p.relative_to(root.parent))

pack_done_results() {
  mkdir -p "$STATE"
  python - <<'PYDONE'

pack_timeout_current_done() {
  [[ -d "$STATE" ]] || return 0
  python - <<'PYTIMEOUTZIP'
import json, math, os, time, zipfile
from pathlib import Path

state = Path(os.environ["STATE"])
dst = Path(os.environ["TIMEOUT_ZIP"])
target = os.environ["EXPERIMENT_ID"]
methods = [x for x in os.environ["METHODS_CSV"].split(",") if x]
seeds = [int(x) for x in os.environ["SEEDS_CSV"].split(",") if x]

complete_methods = []
complete_final_seeds = []
completed_lr_candidates = []
failed_lr_candidates = []

for method in methods:
    method_root = state / "outputs" / method / target / method
    final_seed_ok = []
    for seed in seeds:
        test_summary = method_root / f"seed_{seed}" / "test_summary.json"
        if test_summary.is_file():
            final_seed_ok.append(seed)
            complete_final_seeds.append({"method": method, "seed": seed})
    if len(final_seed_ok) == len(seeds):
        complete_methods.append(method)

    for status_path in method_root.rglob("candidate_status.json") if method_root.exists() else []:
        try:
            row = json.loads(status_path.read_text(encoding="utf-8"))
            rc = int(row.get("return_code", 1))
            val = float(row.get("best_val_acc1", float("nan")))
            item = {
                "method": method,
                "seed": int(row.get("seed", -1)),
                "lr": float(row.get("lr", float("nan"))),
                "return_code": rc,
                "best_val_acc1": val,
                "best_epoch": row.get("best_epoch"),
                "output_dir": row.get("output_dir"),
            }
            if rc == 0 and math.isfinite(val):
                completed_lr_candidates.append(item)
            else:
                failed_lr_candidates.append(item)
        except Exception as exc:
            failed_lr_candidates.append({
                "method": method,
                "status_file": str(status_path),
                "reason": f"unreadable candidate status: {exc}",
            })

manifest = {
    "job": os.environ["JOB"],
    "reason": "11h20_wallclock_guard",
    "created_epoch": int(time.time()),
    "elapsed_seconds": int(time.time()) - int(os.environ["CELL_START_TS"]),
    "budget_seconds": int(os.environ["CELL_BUDGET_SECONDS"]),
    "experiment_id": target,
    "requested_methods": methods,
    "requested_seeds": seeds,
    "complete_methods": complete_methods,
    "complete_final_seed_runs": complete_final_seeds,
    "completed_validation_lr_candidates": completed_lr_candidates,
    "failed_or_nonfinite_lr_candidates": failed_lr_candidates,
    "note": (
        "This timeout export contains every non-checkpoint result/log/config file "
        "available when the 11h20 guard fired. It can therefore contain completed "
        "validation LR candidates even when a whole method has not yet reached final test."
    ),
}

tmp = dst.with_name(dst.name + f".tmp.{os.getpid()}")
tmp.unlink(missing_ok=True)
with zipfile.ZipFile(tmp, "w", zipfile.ZIP_DEFLATED, compresslevel=6) as z:
    z.writestr("11H20_CURRENT_DONE_MANIFEST.json", json.dumps(manifest, indent=2, default=str) + "\n")
    for p in state.rglob("*"):
        if not p.is_file():
            continue
        rel = p.relative_to(state)
        # Result export, not a model archive: omit large model checkpoints.
        if p.suffix.lower() in {".pth", ".pt", ".ckpt"}:
            continue
        # Smoke/plan are preflight artefacts, not experimental results.
        if rel.parts and rel.parts[0] in {"smoke", "plan"}:
            continue
        z.write(p, Path(os.environ["JOB"]) / rel)

tmp.replace(dst)
print("11H20 CURRENT-DONE ZIP READY:", dst)
print("Complete methods:", complete_methods)
print("Complete final seed runs:", complete_final_seeds)
print("Completed LR candidates:", len(completed_lr_candidates))
PYTIMEOUTZIP
  ls -lh "$TIMEOUT_ZIP" || true
}

stop_active_training_groups() {
  [[ -f "$ACTIVE_PGIDS_FILE" ]] || return 0
  mapfile -t _pgids < "$ACTIVE_PGIDS_FILE" || true
  for pgid in "${_pgids[@]:-}"; do
    [[ "$pgid" =~ ^[0-9]+$ ]] || continue
    kill -TERM -- "-$pgid" 2>/dev/null || true
  done
  sleep 20
  for pgid in "${_pgids[@]:-}"; do
    [[ "$pgid" =~ ^[0-9]+$ ]] || continue
    kill -KILL -- "-$pgid" 2>/dev/null || true
  done
}

watchdog_11h20() {
  local now remaining
  now=$(date +%s); remaining=$((CELL_DEADLINE_TS-now))
  if (( remaining > 0 )); then sleep "$remaining"; fi
  mkdir -p "$STATE"
  {
    echo "11h20 wall-clock limit reached"
    echo "cell_start_epoch=$CELL_START_TS"
    echo "deadline_epoch=$CELL_DEADLINE_TS"
    echo "reached_epoch=$(date +%s)"
  } > "$TIMEOUT_FLAG"
  echo "11h20 LIMIT REACHED — stopping active training and exporting."
  stop_active_training_groups
  pack_all_exports || true
  pack_timeout_current_done || true
  sync || true
  echo "11h20 timeout export: $TIMEOUT_ZIP"
}

  trap - EXIT

  pack_all_exports || true
  if [[ -f "$TIMEOUT_FLAG" ]] || [[ "$rc" -eq 124 ]]; then
    pack_timeout_current_done || true
  fi
  exit "$rc"
}

# A failed/non-finite candidate excludes that LR globally, but does

# ------------------------------------------------------------

replacement=r'''def _candidate_failure_reason(row: dict[str, Any]) -> str | None:
    """Return why an LR-search row is unusable, or ``None`` when it is valid.

    A failed hyperparameter trial is not itself a protocol failure.  In
    particular, aggressive LoRA-Conv learning rates can occasionally diverge
    and produce a non-finite loss.  The fair response is to mark that *whole LR*
    ineligible across all seeds, rather than aborting the complete method or
    selecting different LRs per seed.
    """
    try:
        return_code = int(row.get("return_code", 1))
    except (TypeError, ValueError):
        return "invalid_return_code"
    if return_code != 0:
        return f"return_code={return_code}"

    try:
        best_val = float(row.get("best_val_acc1", float("nan")))
    except (TypeError, ValueError):
        return "invalid_best_val_acc1"
    if not math.isfinite(best_val):
        return "nonfinite_best_val_acc1"

    try:
        best_epoch = int(row.get("best_epoch", -1))
    except (TypeError, ValueError):
        return "invalid_best_epoch"
    if best_epoch < 0:
        return "missing_best_epoch"
    return None


def select_shared_lr(
    candidate_rows: list[dict[str, Any]],
    *,
    seeds: list[int],
    lr_candidates: list[float],
) -> dict[str, Any]:
    """Select one shared LR by mean validation accuracy across seeds.

    Every seed/LR pair must still be *attempted*.  If a candidate LR fails or
    yields a non-finite validation result for any seed, that LR is excluded for
    every seed.  Selection is then performed only over LRs that are valid for
    the complete requested seed set.  This keeps the shared-LR protocol intact
    while allowing numerically unstable trials (notably high-LR LoRA-Conv
    trials) to be treated as ordinary failed hyperparameter candidates.
    """
    expected = {(int(seed), float(lr)) for seed in seeds for lr in lr_candidates}
    observed = {(int(row["seed"]), float(row["lr"])) for row in candidate_rows}
    if observed != expected or len(candidate_rows) != len(expected):
        missing = sorted(expected - observed)
        extra = sorted(observed - expected)
        raise RuntimeError(
            "Fair LR search is incomplete or duplicated; "
            f"missing={missing}, extra={extra}, rows={len(candidate_rows)}, expected={len(expected)}"
        )

    aggregates: list[dict[str, Any]] = []
    invalid_candidates: list[dict[str, Any]] = []
    excluded_lrs: list[float] = []

    for lr in lr_candidates:
        rows = sorted(
            [row for row in candidate_rows if float(row["lr"]) == float(lr)],
            key=lambda row: int(row["seed"]),
        )
        failures = []
        for row in rows:
            reason = _candidate_failure_reason(row)
            if reason is not None:
                failure = {
                    "seed": int(row["seed"]),
                    "lr": float(row["lr"]),
                    "reason": reason,
                    "return_code": row.get("return_code"),
                    "best_val_acc1": row.get("best_val_acc1"),
                    "output_dir": row.get("output_dir"),
                }
                failures.append(failure)
                invalid_candidates.append(failure)

        # Shared-LR fairness: an LR is either eligible for all requested seeds
        # or excluded for all requested seeds.  Never cherry-pick per-seed LRs.
        if failures:
            excluded_lrs.append(float(lr))
            continue

        values = [float(row["best_val_acc1"]) for row in rows]
        aggregates.append({
            "lr": float(lr),
            "mean_best_val_acc1": float(statistics.mean(values)),
            "std_best_val_acc1": float(statistics.stdev(values)) if len(values) > 1 else 0.0,
            "per_seed_best_val_acc1": {str(int(row["seed"])): float(row["best_val_acc1"]) for row in rows},
            "per_seed_best_epoch": {str(int(row["seed"])): int(row["best_epoch"]) for row in rows},
        })

    if not aggregates:
        raise RuntimeError(
            "Fair LR search has no candidate that finished successfully with a "
            "finite validation score for every requested seed. "
            f"Failed candidates: {invalid_candidates}"
        )

    selected = max(
        aggregates,
        key=lambda row: (float(row["mean_best_val_acc1"]), -float(row["lr"])),
    )
    return {
        "selection_scope": "method_across_seeds",
        "aggregation": "mean_best_val_acc1",
        "tie_break": "lower_lr",
        "seeds": [int(s) for s in seeds],
        # Keep the original attempted grid for auditability.
        "lr_candidates": [float(lr) for lr in lr_candidates],
        "eligible_lr_candidates": [float(row["lr"]) for row in aggregates],
        "excluded_lr_candidates": excluded_lrs,
        "invalid_candidates": invalid_candidates,
        "failed_candidate_policy": "exclude_lr_globally_if_any_seed_invalid",
        "candidates": aggregates,
        "selected_lr": float(selected["lr"]),
        "selected_mean_best_val_acc1": float(selected["mean_best_val_acc1"]),
    }
'''
